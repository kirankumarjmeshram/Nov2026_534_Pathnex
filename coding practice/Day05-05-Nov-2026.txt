- Ansible - Create user and set permissions

- name: create a new user
  hosts: all
  become: yes
  tasks: 
    - name: pathnex
      user:
        name: pathnex
        state: present
        shell: /bin/bash
    - name: Set permissions for thr user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
